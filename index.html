# ===================== CELDA 1: carga =====================
import pandas as pd
import numpy as np

df = pd.read_csv('adult.csv', sep=";", header=None)          # CSV sin cabecera, separado por ;
df.columns = columns = ["age", "workclass", "fnlwgt", "education", "education_num",
                        "marital_status", "occupation", "relationship", "race", "sex",
                        "capital_gain", "capital_loss", "hours_per_week", "native_country", "income"]
# asigna nombres (el "columns = " intermedio solo guarda además la lista en una variable)

# ===================== CELDA 2: limpieza =====================
df.replace('?', np.nan, inplace=True)                        # '?' -> NaN. inplace=True modifica df directamente

for col in df.columns:
    if df[col].dtype != 'object':                            # columna numérica
        df[col] = df[col].fillna(df[col].mean())             # NaN -> media de la columna
    else:                                                    # columna de texto
        df[col] = df[col].fillna(df[col].mode()[0])          # NaN -> moda (valor más frecuente)

print('Tamaño del dataset después de imputación:', df.shape)
print('Valores faltantes restantes:', df.isna().sum().sum()) # primer sum() por columna, segundo suma total -> debe ser 0

from tensorflow import keras                              # Keras: API de alto nivel de TensorFlow para redes neuronales

# ===================== CELDA 5: cargar datos =====================
(x_train, y_train), (x_test, y_test) = keras.datasets.fashion_mnist.load_data()
# Fashion-MNIST: 60.000 imágenes de entrenamiento y 10.000 de test, en gris 28x28, de 10 tipos de prenda. Se descarga solo

# ===================== CELDA 6: definir el problema =====================
num_classes = 10                  # 10 clases de salida (camiseta, pantalón, jersey...)
input_shape = (28, 28, 1)         # (alto, ancho, canales). 1 canal = escala de grises (RGB sería 3)

# ===================== CELDA 7: arquitectura de la CNN =====================
from tensorflow.keras import layers, models, optimizers

model = models.Sequential([                               # Sequential: capas apiladas una tras otra
    layers.Input(shape=input_shape),                      # capa de entrada con la forma de la imagen

    layers.Conv2D(32, (3, 3), padding="same", activation="relu"),
    # 32 filtros de 3x3 que deslizan por la imagen y detectan patrones (bordes, texturas). padding="same": mantiene el tamaño 28x28
    layers.MaxPooling2D((2, 2)),                          # reduce a la mitad (14x14) quedándose con el máximo de cada bloque 2x2

    layers.Conv2D(64, (3, 3), padding="same", activation="relu"),   # 64 filtros: patrones más complejos
    layers.MaxPooling2D((2, 2)),                          # 14x14 -> 7x7

    layers.Conv2D(128, (3, 3), padding="same", activation="relu"),  # 128 filtros: patrones aún más abstractos
    layers.Dropout(0.3),                                  # apaga al azar el 30% de neuronas en entrenamiento -> evita overfitting

    layers.Flatten(),                                     # aplana el mapa 3D a un vector 1D para pasarlo a capas densas
    layers.Dense(128, activation="relu"),                 # capa totalmente conectada de 128 neuronas
    layers.Dense(num_classes, activation="softmax")       # salida: 10 neuronas; softmax las convierte en probabilidades que suman 1
])

model.compile(
    optimizer=optimizers.Adam(1e-3),                      # Adam con learning rate 0.001
    loss="sparse_categorical_crossentropy",
    # pérdida para multiclase cuando las etiquetas son ENTEROS (0..9). Si fuesen one-hot se usaría categorical_crossentropy
    metrics=["accuracy"]                                  # métrica a mostrar
)

model.summary()                                           # tabla con capas, formas de salida y nº de parámetros

    x_train_norm, y_train,

)   # history guarda la pérdida/accuracy por epoch (útil para graficar)

test_loss, test_acc = model.evaluate(x_test_norm, y_test) # evalúa con el test, que la red NUNCA ha visto
print(f"\nExactitud (Accuracy) en test para la CNN: {test_acc:.4f}")
